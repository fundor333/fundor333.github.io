{{- /* "Also posted on:" — links from .Params.syndication (list of URLs). The visible label is prettified per platform: reddit.com -> r/ bsky.app -> @ .../@user/... -> /@ (Mastodon & friends) otherwise -> (without "www.") The href and the u-syndication / rel="syndication" markup are unchanged. */ -}} {{- with .Params.syndication -}}
{{ T "syndication.also_on" }} {{- range $i, $url := . -}} {{- $u := urls.Parse $url -}} {{- $host := strings.TrimPrefix "www." $u.Host -}} {{- $label := $host -}} {{- if strings.Contains $host "reddit.com" -}} {{- with (index (findRE `/r/[^/?#]+` $u.Path) 0) }}{{ $label = strings.TrimPrefix "/" . }}{{ end -}} {{- else if eq $host "bsky.app" -}} {{- with (index (findRE `/profile/[^/?#]+` $u.Path) 0) }}{{ $label = printf "@%s" (strings.TrimPrefix "/profile/" .) }}{{ end -}} {{- else if hasPrefix $u.Path "/@" -}} {{- with (index (findRE `/@[^/?#]+` $u.Path) 0) }}{{ $label = printf "%s%s" $host . }}{{ end -}} {{- end -}} {{- if $i }}, {{ end -}} {{ $label }} {{- end -}}
{{- end -}}
